<body>

<h2 id="title">My Website</h2>

<p id="message">Hello World</p>
<button onclick="change()">Change Message</button>
<button onclick="color()">Change Color</button>

<p id="date"></p>
<button onclick="showDate()">Show Date</button>

<br><br>

<input type="number" id="num1" placeholder="First number">
<input type="number" id="num2" placeholder="Second number">
<button onclick="add()">Add</button>
<p id="result"></p>

<div id="box" style="background-color: lightgray; padding: 20px;">
    My Box
</div>
<button onclick="changeBox()">Change Box Color</button>

<script>
function change() {
    document.getElementById("message").innerHTML = "Welcome!";
}

function color() {
    document.getElementById("title").style.color = "red";
}

function showDate() {
    document.getElementById("date").innerHTML = new Date();
}

function add() {
    var a = Number(document.getElementById("num1").value);
    var b = Number(document.getElementById("num2").value);
    document.getElementById("result").innerHTML = a + b;
}

function changeBox() {
    document.getElementById("box").style.backgroundColor = "yellow";
}
</script>
